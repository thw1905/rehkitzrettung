// DJI-Pilot-Sitzungen sind zeitlich begrenzt und an einen ausgewaehlten Auftrag gebunden.
model DjiPilotSession {
  id          String   @id @default(cuid())
  tokenHash   String   @unique
  pilotUserId String
  orderId     String
  workspaceId String
  expiresAt   DateTime
  revokedAt   DateTime?
  createdAt   DateTime @default(now())

  @@index([pilotUserId, expiresAt])
}

model DjiMapElement {
  id          String   @id @default(cuid())
  workspaceId String
  elementId   String
  groupId     String
  pilotUserId String
  orderId     String
  name        String
  latitude    Float
  longitude   Float
  resource    Json
  status      String   @default("PENDING")
  pinpointId  String?  @unique
  receivedAt  DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@unique([workspaceId, elementId])
  @@index([pilotUserId, receivedAt])
  @@index([orderId])
}
